## 1. 方案：为什么用这一条主线

### 1.1 五段流水线

```
物理量（辐射度）
   │  ① 渲染方程：定义"要算什么积分"
   ▼
积分目标（方向积分 / 面积积分 / 路径积分）
   │  ② 采样：把积分变成"在哪些点上取值、每个点权重多少"
   ▼
样本（方向、pdf、权重）
   │  ③ 积分：估计器（蒙特卡洛 / 重要性采样 / MIS / RR / 预积分）
   ▼
带噪的数值（像素颜色 / 探针系数 / 簇可见性）
   │  ④ 重建：滤波、插值、降噪、上采样、色调映射
   ▼
像素值（显示）
```

每一段都有"必须守住的量纲与不变量"，上一段的错误会在下一段被放大或掩盖。

## 2. 渲染方程：物理与量纲

### 2.1 辐射度量学的四个量

| 量 | 符号 | 单位 | 含义 |
|---|---|---|---|
| 辐射通量 | $\Phi$ | W | 单位时间能量 |
| 辐照度 | $E$ | W/m² | 单位面积入射功率 |
| 辐射强度 | $I$ | W/sr | 单位立体角功率 |
| **辐射度** | $L$ | W/(m²·sr) | 单位面积、单位立体角；**渲染里最核心的量** |

$$
L = \frac{\mathrm d^2\Phi}{\mathrm dA\cos\theta\,\mathrm d\omega}
$$

### 2.2 表面渲染方程

$$
L_o(x,\omega_o) = L_e(x,\omega_o) + \int_{\Omega} f_r(x,\omega_i,\omega_o)\,L_i(x,\omega_i)\,\cos\theta_i\,\mathrm d\omega_i
$$

### 2.3 BRDF：Cook–Torrance

引擎的 PBR 分解（`pbr_common.slang`）：

```slang
// 摘自 Engine/Shader/Shaders/pbr_common.slang:22-40
// Fresnel — Schlick 近似
float3 F_Schlick(float3 f0, float cosTheta) {
    return f0 + (1.0 - f0) * pow(clamp(1.0 - cosTheta, 0.0, 1.0), 5.0);
}

// Normal Distribution Function — GGX (Trowbridge-Reitz)
float D_GGX(float NdotH, float roughness) {
    float alpha = roughness * roughness;
    float alpha2 = alpha * alpha;
    float denom = NdotH * NdotH * (alpha2 - 1.0) + 1.0;
    return alpha2 / (HE_PI * denom * denom);
}
```

```slang
// 摘自 pbr_common.slang:59-81
// Geometry Function — Smith + Schlick-GGX
float G_SchlickGGX(float NdotV, float roughness) {
    float k = (roughness + 1.0) * (roughness + 1.0) / 8.0;
    return NdotV / (NdotV * (1.0 - k) + k);
}

// 【两套 Smith 的 k 不可混用】本函数用 Karis/UE4 的**直接光**惯例 k = (roughness+1)²/8；
// IBL 烘焙端用的是另一套 k = α/2（见下方 G_Smith_IBL，split-sum 的 LUT 必须与预滤波同族）。
float G_Smith_Direct(float NdotV, float NdotL, float roughness) {
    return G_SchlickGGX(NdotV, roughness) * G_SchlickGGX(NdotL, roughness);
}

// IBL 端的 Smith-GGX：k = α/2（α = roughness²），与 `IBL_BRDF_LUT.frag` 的积分同族。
float G_Smith_IBL(float NdotV, float NdotL, float roughness) {
    float a = roughness * roughness;   // α（Disney/GGX 的 α = roughness²）
    float k = a * 0.5;                 // IBL 简化 k = α/2
    float G1V = NdotV / (NdotV * (1.0 - k) + k);
    float G1L = NdotL / (NdotL * (1.0 - k) + k);
    return G1V * G1L;
}
```

## 3. 采样：把积分变成有限次实验

### 3.1 概率密度与估计量

$$
\hat I = \frac{1}{N}\sum_{i=1}^{N}\frac{f(x_i)}{p(x_i)},\qquad x_i \sim p
$$

- **无偏**：$\mathbb E[\hat I]=I$；
- **方差**：$\mathrm{Var}(\hat I) \propto \frac{1}{N}\left(\frac{f}{p}\text{ 的方差}\right)$。

提高 $N$ 更贵，让 $p$ 接近 $f$（重要性采样）更划算。

### 3.2 重要性采样：代码

**余弦加权半球**（漫反射）：

```slang
// 摘自 Engine/Shader/Shaders/Sampling.slang:15-26
// 余弦权重半球采样（绕法线 N），pdf = NdotL / PI
float3 CosineSampleHemisphere(float3 N, float2 u) {
    float phi = 2.0 * 3.14159265 * u.x;
    float r = sqrt(u.y);
    float x = r * cos(phi);
    float y = r * sin(phi);
    float z = sqrt(max(0.0, 1.0 - u.y));
    float3 Up = abs(N.z) < 0.999 ? float3(0, 0, 1) : float3(1, 0, 0);
    float3 T = normalize(cross(Up, N));
    float3 B = cross(N, T);
    return normalize(x * T + y * B + z * N);
}
```

**GGX 半向量采样**（高光）：

```slang
// 摘自 Sampling.slang:62-80
float3 SampleGgxHalf(float3 N, float roughness, float2 u, float minAlpha,
                     out float NdotH) {
    float alpha = max(roughness * roughness, minAlpha);
    float cosTheta = sqrt((1.0 - u.y) / max(1.0 + (alpha * alpha - 1.0) * u.y, 1e-6));
    float sinTheta = sqrt(max(0.0, 1.0 - cosTheta * cosTheta));
    float phi = 6.28318530718 * u.x;
    float3 H = float3(sinTheta * cos(phi), sinTheta * sin(phi), cosTheta);
    float3 Up = abs(N.z) < 0.999 ? float3(0, 0, 1) : float3(1, 0, 0);
    float3 T = normalize(cross(Up, N));
    float3 B = cross(N, T);
    NdotH = cosTheta;
    return normalize(H.x * T + H.y * B + H.z * N);
}
```

### 3.3 低差异序列

```slang
// 摘自 Sampling.slang:28-45
float2 Hammersley(uint i, uint N) {
    return float2(float(i) / float(N), RadicalInverse_VdC(i));
}

float RadicalInverse_VdC(uint bits) {
    bits = (bits << 16u) | (bits >> 16u);
    bits = ((bits & 0x55555555u) << 1u) | ((bits & 0xAAAAAAAAu) >> 1u);
    bits = ((bits & 0x33333333u) << 2u) | ((bits & 0xCCCCCCCCu) >> 2u);
    bits = ((bits & 0x0F0F0F0Fu) << 4u) | ((bits & 0xF0F0F0F0u) >> 4u);
    bits = ((bits & 0x00FF00FFu) << 8u) | ((bits & 0xFF00FF00u) >> 8u);
    return float(bits) * 2.3283064365386963e-10;
}
```

### 3.5 随机数：PCG

```slang
// 摘自 Engine/Shader/Shaders/Lumen/ScreenProbeSampling.slang:23-30
// 确定性的低差异随机（PCG 变体）：同一 (探针, 光线, 帧) 恒得同一方向 ⇒ 可复现
uint Pcg(uint v) {
    v = v * 747796405u + 2891336453u;
    const uint w = ((v >> ((v >> 28u) + 4u)) ^ v) * 277803737u;
    return (w >> 22u) ^ w;
}
float U01(uint seed) { return (float)(Pcg(seed) & 0x00FFFFFFu) / 16777216.0; }
```

## 4. 积分：从样本到数值

### 4.1 蒙特卡洛积分与收敛

$$
\hat I_N = \frac1N\sum_{i=1}^{N}\frac{f(x_i)}{p(x_i)},\qquad
\text{误差} \sim \frac{\sigma}{\sqrt N}
$$

### 4.2 NEE 与 MIS：代码

```slang
// 摘自 Engine/Shader/Shaders/PathTracing/PT_Full.rgen.slang:184-206（节选）
if ((g_PC.flags & PT_FLAG_NEE) != 0u && !restirDone) {
    // NEE：遍历全部光源 + 阴影射线 + MIS
    for (uint li = 0; li < min(g_PC.lightCount, PT_MAX_LIGHTS); li++) {
        float3 lightPos2, L; float3 Li; float lightPdf;
        if (!SampleLight(g_Lights[li], P, lightPos2, L, Li, lightPdf)) continue;
        float ndl = dot(N, L);
        if (ndl <= 0.0) continue;

        float3 brdf = PT_EvalBRDF(mat, N, V, L);

        // BRDF 采样 pdf：直接调用采样端的同一函数（三瓣混合分布），
        // MIS 权重引用的密度因此与 SampleBSDF 实际使用的策略严格一致
        float brdfPdf = (1.0 - transmission) * BSDFPdf(mat, N, V, L);

        float weight = 1.0;
        if ((g_PC.flags & PT_FLAG_MIS) != 0u)
            weight = PowerHeuristic(lightPdf, brdfPdf);

        float vis = TraceShadowRay(P, N, L, length(lightPos2 - P));
        radiance += throughput * brdf * Li * vis * weight / lightPdf;
    }
}
```

### 4.4 预积分：IBL split-sum

```slang
// 摘自 Engine/Shader/Shaders/GI/IBL_Prefilter.frag.slang:20-42
static const uint SAMPLE_COUNT = 256;

float4 fragmentMain(VSOutputWorldDir input) : SV_Target0 {
    float3 N = normalize(input.worldDir);
    float3 V = N;  // 对于 cubemap: viewing direction = reflection direction = normal

    float3 color = float3(0.0);
    float  totalWeight = 0.0;

    for (uint i = 0; i < SAMPLE_COUNT; i++) {
        float2 Xi = Hammersley(i, SAMPLE_COUNT);
        float3 H  = SampleGgxHalf(N, roughness, Xi, 0.0);   // 共享采样器
        float3 L  = normalize(2.0 * dot(V, H) * H - V);     // 反射方向

        float NdotL = max(dot(N, L), 0.0);
        if (NdotL > 0.0) {
            color += u_SkyboxTexture.SampleLevel(u_SkyboxSampler, L, 0.0).rgb * NdotL;
            totalWeight += NdotL;
        }
    }
    color = totalWeight > 0.0 ? color / totalWeight : float3(0.0);
    return float4(color, 1.0);
}
```

```slang
float3 N    = normalize(u_Normal.Sample(u_PointSampler, uv).rgb * 2.0 - 1.0);
bool offScreen = historyUV.x < 0.0 || historyUV.x > 1.0 ||
                 historyUV.y < 0.0 || historyUV.y > 1.0;
float disocclusion = 0.0;
if (!offScreen) {
    float depthHist = u_Depth.Sample(u_PointSampler, historyUV).r;
    float3 N_hist   = normalize(u_Normal.Sample(u_PointSampler, historyUV).rgb * 2.0 - 1.0);
    float depthDiff = abs(depth - depthHist) / max(depth, 0.001);
    disocclusion = max(disocclusion, smoothstep(0.001, 0.01, depthDiff));
    float normalDiff = 1.0 - dot(N, N_hist);
    disocclusion = max(disocclusion, smoothstep(0.001, 0.05, normalDiff));
    float speed = length(vel);
    disocclusion = max(disocclusion, smoothstep(0.01, 0.05, speed));
} else {
    disocclusion = 1.0;
}

// 4. 混合
if (disocclusion > 0.95 || offScreen) {
    output = float4(currColor, 1.0);
} else {
    float3 histColor = u_HistoryColor.Sample(u_PointSampler, historyUV).rgb;
    // YCoCg 邻域 AABB 裁剪
    float3 aabbMin, aabbMax;
    current_color_aabb(uv, aabbMin, aabbMax);
    float3 histYC = rgb_to_ycocg(histColor);
    histYC = clamp(histYC, aabbMin, aabbMax);
    float3 clampedHist = ycocg_to_rgb(histYC);
    float blend = lerp(u_TAA.blendFactor, 0.3, disocclusion);
    resolved = lerp(clampedHist, currColor, blend);
}
```

### 5.2 降噪：À-Trous 小波

```slang
// 摘自 Engine/Shader/Shaders/PathTracing/PT_Atrous.comp.slang:43-85（节选）
for (int dy = -1; dy <= 1; dy++)
for (int dx = -1; dx <= 1; dx++) {
    float l = Luminance(g_Color.Load(int3(int2(p) + int2(dx, dy), 0)).rgb);
    mean += l; meanSq += l * l;
}
mean /= 9.0; meanSq /= 9.0;
float sigmaColor = g_PC.sigmaColor * sqrt(max(meanSq - mean * mean, 1e-6));

float3 sum = centerC;
float  wSum = 1.0;
float  step = 1.0;
for (uint i = 0; i < iters; i++) {
    int2 s = int2(step);
    int2 offs[4] = { int2( s.x, 0), int2(-s.x, 0), int2(0,  s.y), int2(0, -s.y) };
    for (uint k = 0; k < 4; k++) {
        int2 np = int2(p) + offs[k];
        float3 c = g_Color.Load(int3(np, 0)).rgb;
        float  z = g_Depth.Load(int3(np, 0)).r;
        float3 n = normalize(g_Normal.Load(int3(np, 0)).xyz * 2.0 - 1.0);

        float wDepth = exp(-abs(z - centerZ) / max(g_PC.sigmaDepth, 1e-4));
        float wNorm  = pow(max(dot(n, centerN), 0.0), g_PC.normalPower);
        float wCol   = exp(-abs(Luminance(c) - centerLum) / max(sigmaColor, 1e-4));
        float w = wDepth * wNorm * wCol;

        sum += c * w;
        wSum += w;
    }
    step *= 2.0;
}
g_Output[p] = float4(sum / max(wSum, 1e-4), 1.0);
```

### 5.4 几何重建：重心坐标与透视校正

```slang
// 摘自 Engine/Shader/Shaders/Nanite/Nanite_SoftRasterCommon.slang:457-505（节选）
bool softRasterCovered(SoftRasterTriangle tri, float2 p, out float3 bary) {
    float2 e0 = tri.p1 - tri.p0;
    float2 e1 = tri.p2 - tri.p1;
    float2 e2 = tri.p0 - tri.p2;
    // 子面积于是写成 `e.x*(p.y-pk.y) - e.y*(p.x-pk.x)`，**顺序不能反**：
    // 写反会让子面积与总面积差一个负号，结果"三角形内部的三个边函数全为负" ⇒ 一个像素都覆盖不到
    float a0 = e0.x * (p.y - tri.p0.y) - e0.y * (p.x - tri.p0.x);
    float a1 = e1.x * (p.y - tri.p1.y) - e1.y * (p.x - tri.p1.x);
    float a2 = e2.x * (p.y - tri.p2.y) - e2.y * (p.x - tri.p2.x);

    // 【双面光栅化：必须按面积符号翻转】边函数带"手性"：
    // 屏幕空间里顺时针绕序的三角形，三个边函数在三角形内部**全为负**
    float area2 = (tri.p1.x - tri.p0.x) * (tri.p2.y - tri.p0.y)
                - (tri.p2.x - tri.p0.x) * (tri.p1.y - tri.p0.y);
    float flip = (area2 < 0.0) ? -1.0 : 1.0;
    a0 *= flip; a1 *= flip; a2 *= flip;

    const float kEdgeTolerance = -1.0e-4;
    if (a0 < kEdgeTolerance || a1 < kEdgeTolerance || a2 < kEdgeTolerance) {
        bary = float3(0.0, 0.0, 0.0); return false;
    }
    float sum = a0 + a1 + a2;
    if (!(abs(sum) > 1.0e-12)) { bary = float3(0.0, 0.0, 0.0); return false; }
    // 面积权重：a1 对顶点 0 的权重、a2 对顶点 1、a0 对顶点 2
    bary = float3(a1 / sum, a2 / sum, a0 / sum);
    return true;
}

// 透视校正插值：属性 = Σ(w_k × attr_k) / Σ(w_k)，其中 w_k = bary_k / clipW_k
float3 softRasterAttribute(float3 bary, SoftRasterTriangle tri, float3 a0, float3 a1, float3 a2) {
    float w0 = bary.x * tri.invW0;
    float w1 = bary.y * tri.invW1;
    float w2 = bary.z * tri.invW2;
    float sum = w0 + w1 + w2;
    if (!(abs(sum) > 1.0e-20)) return a0;
    return (a0 * w0 + a1 * w1 + a2 * w2) / sum;
}
```

```cpp
    auto makePlane = [&](u32 rowN, bool add) -> float4 {
```

## 7. 不变量与常见错误

### 7.1 量纲：$E$ 与 $L$ 差一个 $1/\pi$

```slang
// 摘自 Engine/Shader/Shaders/Lighting/DeferredLighting.frag.slang:210-213
case GISOURCE_DDGI:
    // EvalDDGI_SH 已施加 Lambert 卷积 A_l → 返回辐照度 E → 需乘 albedo/π
    return SampleDDGI(worldPos, N) * albedo * (1.0 / HE_PI);
```

整条链路里 $1/\pi$ 只应出现一次：多补或少补都是 $\pi$ 倍误差。

### 7.2 pdf 必须归一化

```slang
// 摘自 Engine/Shader/Shaders/GI/DDGI.comp.slang:234-237
// 归一化：蒙特卡洛积分缩放因子 4π/N × 强度
float scale = (4.0 * 3.14159265359) / validSamples * u_Params.x;
```

半球均匀采样是 $2\pi/N$（`Lumen_ScreenProbe_SHProject.comp.slang:82`），整球是 $4\pi/N$。

### 7.3 白炉判据

```cpp
// 摘自 Engine/Render/Lumen/LumenSH.h:25-26
/// 白炉下 0 阶系数的解析值：2π·Y00 = √π
inline constexpr double kSHWhiteFurnaceL0 = 1.7724538509055159;   // √π
```

白炉（入射辐射度恒为 1）下，所有估计器都应得到 $E\equiv\pi$（或 $L\equiv1$）。
这是**不需要参考图**的解析验收。

### 7.4 确定性与精度

- GPU 原子/浮点累加顺序不定 → 需要确定性时用定点/固定顺序（Nanite 深度键、SH 统计定点）；
- 量化压缩（16bit 位置、10bit 法线）会引入误差，需要重新标定判据；
- 低精度（fp16）在 HDR 链路上要小心溢出。

### 7.5 常见 bug 清单

| 症状 | 常见原因 | 检查点 |
|---|---|---|
| 画面整体偏亮/偏暗 $\pi$ 倍 | 量纲错误 | §7.1 |
| 噪声大、收敛慢 | 采样分布与积分不匹配 | §3.2 |
| 画面"发灰" | 色调映射/曝光/色彩空间 | §5.5 |
