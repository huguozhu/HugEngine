# Vulkan validation layer settings used by the GI verification runs.
# Purpose: lift the duplicate-message cap so the reported count is the REAL number of
# validation messages (docs section 11.3 / methodology item 6 in 11.3.1). Without this the
# layer collapses repeats and the count depends on message ordering, which makes
# "same as baseline" meaningless.
#
# Usage (VK_LAYER_SETTINGS_PATH points at the DIRECTORY holding this file):
#   $env:VK_LAYER_SETTINGS_PATH = "Tools\gi"
#   powershell -File Tools/gi/dump_gi.ps1 -Frame 120
#
# Reference counts on the three diffuse-stack variants at frame 120:
#   with this file (dedup off)  : none 104 / ddgi 104 / ssgi 110
#   without this file (dedup on): none  56 / ddgi  56 / ssgi  56
# Earlier, before the sampling added the specular/AO dump targets, the same two rows were
#   75 / 75 / 81  and  49 / 49 / 51. The counts are therefore sensitive to the dump target
# set as well as to dedup. Only a same-harness before/after comparison of two binaries is
# meaningful.
khronos_validation.duplicate_message_limit = 0
